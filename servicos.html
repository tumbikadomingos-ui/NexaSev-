let currentFilter='Todos'; let providersLoaded=false; let totalProviders=0; function choose(cat){ document.querySelectorAll('.filter').forEach(x=>x.classList.remove('active')); const btn=[...document.querySelectorAll('.filter')].find(x=>x.textContent===cat); if(btn)btn.classList.add('active'); currentFilter=cat; const search=document.getElementById('searchInput'); if(search)search.value=''; filterCards(); document.getElementById('professionals').scrollIntoView({behavior:'smooth'}); } function setFilter(cat,btn){ currentFilter=cat; document.querySelectorAll('.filter').forEach(x=>x.classList.remove('active')); btn.classList.add('active'); filterCards(); } function normalizeText(value){ return String(value||'') .normalize('NFD') .replace(/[\u0300-\u036f]/g,'') .toLowerCase() .trim(); } function filterCards(){ const input=document.getElementById('searchInput'); const q=normalizeText(input?input.value:''); const active=normalizeText(currentFilter); let visible=0; document.querySelectorAll('#cards .card').forEach(c=>{ const text=normalizeText( (c.dataset.search||'')+' '+ (c.dataset.cat||'')+' '+ (c.innerText||'') ); const okCat=active==='todos'||normalizeText(c.dataset.cat)===active; const okSearch=!q||text.includes(q); const show=okCat&&okSearch; c.hidden=!show; if(show)visible++; }); const empty=document.getElementById('empty'); const grid=document.getElementById('cards'); if(!providersLoaded){ empty.hidden=true; return; } if(totalProviders===0){ empty.textContent='Ainda não existem profissionais publicados. Volte mais tarde!'; empty.hidden=false; }else{ empty.textContent='Nenhum profissional encontrado. Tente outra pesquisa ou categoria.'; empty.hidden=visible>0; } grid.hidden=totalProviders===0||visible===0; } function showLoading(){ document.getElementById('loading').hidden=false; document.getElementById('error').hidden=true; document.getElementById('empty').hidden=true; document.getElementById('cards').hidden=true; } async function getProviders(){ if(!supabaseClient){ throw new Error('A biblioteca do Supabase não foi carregada.'); } const {data,error}=await supabaseClient .from('providers') .select('id,provider_id,category,title,description,location,whatsapp,is_featured,created_at') .order('is_featured',{ascending:false}) .order('created_at',{ascending:false}); if(error)throw error; return data||[]; } /* Procura a primeira fotografia na pasta do profissional. Os vídeos são ignorados para não aparecerem como capa do cartão. */ async function getProviderPhoto(providerId){ if(!providerId||!supabaseClient)return null; const {data,error}=await supabaseClient.storage .from(MEDIA_BUCKET) .list(providerId,{limit:1000,sortBy:{column:'name',order:'desc'}}); if(error||!data)return null; const photo=data.find(file=> file.name && /\.(jpg|jpeg|png|webp|gif|avif)$/i.test(file.name) ); if(!photo)return null; const path=providerId+'/'+photo.name; const {data:publicData}=supabaseClient.storage .from(MEDIA_BUCKET) .getPublicUrl(path); return publicData?.publicUrl||null; } async function renderProviders(){ const grid=document.getElementById('cards'); const loading=document.getElementById('loading'); const errorBox=document.getElementById('error'); const empty=document.getElementById('empty'); providersLoaded=false; totalProviders=0; grid.innerHTML=''; showLoading(); try{ const providers=await getProviders(); /* Carrega as capas em paralelo. Se uma pasta não tiver foto, o cartão continua a funcionar com o ícone padrão. */ const photos=await Promise.all( providers.map(p=>getProviderPhoto(p.provider_id)) ); providers.forEach((p,index)=>{ const article=document.createElement('article'); article.className='card user-provider'; article.dataset.cat=p.category||''; article.dataset.search= (p.title||'')+' '+ (p.category||'')+' '+ (p.location||'')+' '+ (p.description||'')+' '+ (p.whatsapp||''); const phone=String(p.whatsapp||'').replace(/\D/g,'').replace(/^244/,''); const name=p.title||'Profissional'; const cat=p.category||'Serviços'; const loc=p.location||'Angola'; const desc=p.description||'Serviço profissional.'; const badge=p.is_featured?'⭐ Perfil em destaque':'✓ Novo profissional'; const wa=phone ? `WhatsApp` : ''; article.innerHTML=`
👤
${escapeHtml(cat)}

${escapeHtml(name)}

📍 ${escapeHtml(loc)}
${escapeHtml(desc)}
${badge}
${wa}
`; const cover=article.querySelector('.cover'); if(photos[index]){ const img=document.createElement('img'); img.src=photos[index]; img.alt='Fotografia de '+name; img.loading='lazy'; img.onerror=()=>{img.remove();cover.textContent='👤'}; cover.textContent=''; cover.appendChild(img); } grid.appendChild(article); }); totalProviders=providers.length; providersLoaded=true; loading.hidden=true; errorBox.hidden=true; grid.hidden=providers.length===0; empty.hidden=true; filterCards(); }catch(err){ console.error('Erro ao carregar profissionais:',err); providersLoaded=false; loading.hidden=true; grid.hidden=true; empty.hidden=true; errorBox.hidden=false; } } function escapeHtml(s){ return String(s).replace(/[&<>"']/g,m=>({ '&':'&', '<':'<', '>':'>', '"':'"', "'":''' }[m])); } renderProviders();
